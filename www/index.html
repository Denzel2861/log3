create table if not exists public.logbooks (
  user_id uuid primary key references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);

grant usage on schema public to authenticated;
grant select, insert, update, delete on table public.logbooks to authenticated;

alter table public.logbooks enable row level security;

create policy "Users can read own logbook"
  on public.logbooks for select
  to authenticated
  using (auth.uid() = user_id);

create policy "Users can insert own logbook"
  on public.logbooks for insert
  to authenticated
  with check (auth.uid() = user_id);

create policy "Users can update own logbook"
  on public.logbooks for update
  to authenticated
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
